let data = []; function parseMail(){ let text = document.getElementById("mailtext").value; let dateText = (text.match(/訪問日時：(.+)/) || ["",""])[1]; let customer = (text.match(/お客様　：(.+)/) || ["",""])[1]; let staff = (text.match(/営業担当：(.+)/) || ["",""])[1]; // 日付をISO形式へ変換（並び替え用） let dateMatch = dateText.match(/(\d{4})\/(\d{2})\/(\d{2})/); let dateObj = dateMatch ? new Date(`${dateMatch[1]}-${dateMatch[2]}-${dateMatch[3]}`) : null; let record = { rawDate: dateText, date: dateObj, customer: customer, staff: staff }; data.push(record); display(); document.getElementById("mailtext").value = ""; } function display(showToday=false){ let today = new Date(); today.setHours(0,0,0,0); // ✅ 過去削除 ＆ フィルタ let filtered = data.filter(d => { if(!d.date) return false; let dDate = new Date(d.date); dDate.setHours(0,0,0,0); if(showToday){ return dDate.getTime() === today.getTime(); } else { return dDate >= today; } }); // ✅ 日付順ソート filtered.sort((a,b) => a.date - b.date); let html = ""; filtered.forEach(d => { html += `
${d.rawDate}
${d.customer}
担当：${d.staff}
`; }); document.getElementById("list").innerHTML = html; }
